CREATE TABLE IF NOT EXISTS negocios (
    id CHAR(36) PRIMARY KEY,
    nombre VARCHAR(255) NOT NULL,
    tipo_negocio VARCHAR(100) NOT NULL DEFAULT 'reposteria',
    timezone VARCHAR(100) NOT NULL DEFAULT 'America/Argentina/Cordoba',
    moneda VARCHAR(10) NOT NULL DEFAULT 'ARS',
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS clientes (
    id CHAR(36) PRIMARY KEY,
    negocio_id CHAR(36) NOT NULL,
    nombre VARCHAR(255) NOT NULL,
    telefono VARCHAR(50),
    email VARCHAR(255),
    direccion VARCHAR(255),
    notas TEXT,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (negocio_id)
        REFERENCES negocios(id)
        ON DELETE CASCADE
);

-- 3. MATERIAS PRIMAS
CREATE TABLE IF NOT EXISTS materias_primas (
    id CHAR(36) PRIMARY KEY,
    negocio_id CHAR(36) NOT NULL,
    nombre VARCHAR(255) NOT NULL,
    categoria VARCHAR(100) DEFAULT 'Insumo',
    stock_actual DECIMAL(14,3) NOT NULL DEFAULT 0,
    stock_minimo DECIMAL(14,3) NOT NULL DEFAULT 0,
    costo_unitario DECIMAL(14,2) NOT NULL DEFAULT 0,
    unidad VARCHAR(50) NOT NULL DEFAULT 'g',
    proveedor VARCHAR(255),
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (negocio_id)
        REFERENCES negocios(id)
        ON DELETE CASCADE
);

-- 4. RECETAS
CREATE TABLE IF NOT EXISTS recetas (
    id CHAR(36) PRIMARY KEY,
    negocio_id CHAR(36) NOT NULL,
    nombre VARCHAR(255) NOT NULL,
    categoria VARCHAR(100) NOT NULL DEFAULT 'Tortas',
    rendimiento DECIMAL(14,2) NOT NULL DEFAULT 1,
    unidad_rendimiento VARCHAR(100) NOT NULL DEFAULT 'porciones',
    precio_venta DECIMAL(14,2) NOT NULL DEFAULT 0,
    instrucciones TEXT,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (negocio_id)
        REFERENCES negocios(id)
        ON DELETE CASCADE
);

-- 5. INGREDIENTES DE RECETAS
CREATE TABLE IF NOT EXISTS receta_ingredientes (
    id CHAR(36) PRIMARY KEY,
    receta_id CHAR(36) NOT NULL,
    materia_prima_id CHAR(36) NOT NULL,
    cantidad DECIMAL(14,3) NOT NULL,
    unidad VARCHAR(50) NOT NULL,

    FOREIGN KEY (receta_id)
        REFERENCES recetas(id)
        ON DELETE CASCADE,

    FOREIGN KEY (materia_prima_id)
        REFERENCES materias_primas(id)
        ON DELETE CASCADE
);

-- 6. PEDIDOS
CREATE TABLE IF NOT EXISTS pedidos (
    id CHAR(36) PRIMARY KEY,
    negocio_id CHAR(36) NOT NULL,
    cliente_id CHAR(36),
    receta_id CHAR(36),
    cantidad DECIMAL(14,2) NOT NULL DEFAULT 1,
    precio_unitario DECIMAL(14,2) NOT NULL,
    total DECIMAL(14,2) NOT NULL,
    costo_produccion DECIMAL(14,2) NOT NULL DEFAULT 0,
    utilidad DECIMAL(14,2) NOT NULL DEFAULT 0,
    estado VARCHAR(30) NOT NULL,
    fecha_pedido DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_entrega DATETIME NULL,
    fecha_entregado DATETIME NULL,
    stock_descontado BOOLEAN NOT NULL DEFAULT FALSE,
    observaciones TEXT,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (negocio_id)
        REFERENCES negocios(id)
        ON DELETE CASCADE,

    FOREIGN KEY (cliente_id)
        REFERENCES clientes(id)
        ON DELETE SET NULL,

    FOREIGN KEY (receta_id)
        REFERENCES recetas(id)
        ON DELETE RESTRICT
);

-- 7. MOVIMIENTOS DE STOCK
CREATE TABLE IF NOT EXISTS movimientos_stock (
    id CHAR(36) PRIMARY KEY,
    negocio_id CHAR(36) NOT NULL,
    materia_prima_id CHAR(36) NOT NULL,
    tipo VARCHAR(20) NOT NULL,
    cantidad DECIMAL(14,3) NOT NULL,
    motivo TEXT,
    pedido_id CHAR(36),
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (negocio_id)
        REFERENCES negocios(id)
        ON DELETE CASCADE,

    FOREIGN KEY (materia_prima_id)
        REFERENCES materias_primas(id)
        ON DELETE CASCADE,

    FOREIGN KEY (pedido_id)
        REFERENCES pedidos(id)
        ON DELETE SET NULL
);

-- 8. COMPRAS
CREATE TABLE IF NOT EXISTS compras (
    id CHAR(36) PRIMARY KEY,
    negocio_id CHAR(36) NOT NULL,
    materia_prima_id CHAR(36) NOT NULL,
    proveedor VARCHAR(255),
    cantidad DECIMAL(14,3) NOT NULL,
    unidad VARCHAR(50) NOT NULL,
    costo_total DECIMAL(14,2) NOT NULL,
    fecha DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (negocio_id)
        REFERENCES negocios(id)
        ON DELETE CASCADE,

    FOREIGN KEY (materia_prima_id)
        REFERENCES materias_primas(id)
        ON DELETE CASCADE
);

-- 9. GASTOS FIJOS
CREATE TABLE IF NOT EXISTS gastos_fijos (
    id CHAR(36) PRIMARY KEY,
    negocio_id CHAR(36) NOT NULL,
    concepto VARCHAR(255) NOT NULL,
    monto DECIMAL(14,2) NOT NULL,
    periodicidad VARCHAR(50) NOT NULL DEFAULT 'mensual',
    fecha DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (negocio_id)
        REFERENCES negocios(id)
        ON DELETE CASCADE
);

-- ============================================
-- ÍNDICES
-- ============================================

CREATE INDEX IF NOT EXISTS idx_pedidos_negocio_fecha
ON pedidos(negocio_id, fecha_entregado, estado);

CREATE INDEX IF NOT EXISTS idx_pedidos_negocio_estado
ON pedidos(negocio_id, estado);

CREATE INDEX IF NOT EXISTS idx_compras_negocio_fecha
ON compras(negocio_id, fecha);

CREATE INDEX IF NOT EXISTS idx_movimientos_negocio
ON movimientos_stock(negocio_id, materia_prima_id);

CREATE INDEX IF NOT EXISTS idx_recetas_negocio
ON recetas(negocio_id);

CREATE INDEX IF NOT EXISTS idx_clientes_negocio
ON clientes(negocio_id);

CREATE INDEX IF NOT EXISTS idx_materias_primas_negocio
ON materias_primas(negocio_id);

SELECT nombre AS producto, precio_venta AS precio, rendimiento AS porciones
FROM recetas;

-- A4. DISTINCT: categorías de recetas sin repetir
SELECT DISTINCT categoria FROM recetas;

-- A5. Cálculo en el SELECT: precio por porción/unidad
SELECT nombre, precio_venta, rendimiento,
       ROUND(precio_venta / rendimiento, 2) AS precio_por_unidad

-- B1. Comparación simple
SELECT nombre, precio_venta FROM recetas WHERE precio_venta > 20000;

-- B2. Igualdad con texto
SELECT id, total, fecha_pedido FROM pedidos WHERE estado = 'entregado';

-- B3. AND: pedidos entregados de más de 30000
SELECT id, total, estado FROM pedidos WHERE estado = 'entregado' AND total > 30000;

-- B4. OR
SELECT id, estado, fecha_entrega FROM pedidos
WHERE estado = 'pendiente' OR estado = 'en_preparacion';

-- B5. IN (equivale al OR anterior, más corto)
SELECT id, estado, fecha_entrega FROM pedidos
WHERE estado IN ('pendiente', 'en_preparacion');

-- B6. NOT / distinto
SELECT id, estado, total FROM pedidos WHERE estado <> 'cancelado';

-- B7. BETWEEN con fechas: pedidos de septiembre 2026
SELECT id, fecha_pedido, total FROM pedidos
WHERE fecha_pedido BETWEEN '2026-09-01 00:00:00' AND '2026-09-30 23:59:59';

-- B8. LIKE: clientes cuyo nombre empieza con L
SELECT nombre, telefono FROM clientes WHERE nombre LIKE 'L%';

-- B9. LIKE: materias primas que contienen "ch" en el nombre
SELECT nombre, categoria FROM materias_primas WHERE nombre LIKE '%ch%';

-- B10. IS NULL: clientes sin email
SELECT nombre, telefono FROM clientes WHERE email IS NULL;

-- B11. IS NULL: pedidos que todavía no fueron entregados
SELECT id, estado, fecha_entrega FROM pedidos WHERE fecha_entregado IS NULL;

SELECT id, estado FROM pedidos WHERE stock_descontado = TRUE;


-- C1. Recetas de la más cara a la más barata
SELECT nombre, precio_venta FROM recetas ORDER BY precio_venta DESC;

-- C2. Las 3 recetas más caras
SELECT nombre, precio_venta FROM recetas ORDER BY precio_venta DESC LIMIT 3;

-- C3. Orden por dos columnas: categoría A-Z y dentro de cada una por precio
SELECT categoria, nombre, precio_venta FROM recetas ORDER BY categoria ASC, precio_venta DESC;

-- C4. Los 5 pedidos más recientes
SELECT id, fecha_pedido, total FROM pedidos ORDER BY fecha_pedido DESC LIMIT 5;

-- D1. COUNT: cuántos pedidos hay
SELECT COUNT(*) AS cantidad_pedidos FROM pedidos;

-- D2. COUNT con WHERE: cuántos están entregados
SELECT COUNT(*) AS entregados FROM pedidos WHERE estado = 'entregado';

-- D3. MAX, MIN y AVG del precio de las recetas
SELECT MAX(precio_venta)          AS receta_mas_cara,
       MIN(precio_venta)          AS receta_mas_barata,
       ROUND(AVG(precio_venta),2) AS precio_promedio
FROM recetas;

-- D4. SUM, AVG, MAX y MIN sobre pedidos entregados
SELECT SUM(total)              AS ventas_totales,
       SUM(utilidad)           AS utilidad_total,
       ROUND(AVG(total), 2)    AS ticket_promedio,
       MAX(total)              AS pedido_mas_grande,
       MIN(total)              AS pedido_mas_chico
FROM pedidos
WHERE estado = 'entregado';

-- D5. MAX con subconsulta: ¿cuál fue el pedido más caro? (muestra toda la fila)
SELECT id, cliente_id, total, fecha_pedido
FROM pedidos
WHERE total = (SELECT MAX(total) FROM pedidos);

-- D6. MIN y MAX de fechas: primer y último pedido
SELECT MIN(fecha_pedido) AS primer_pedido, MAX(fecha_pedido) AS ultimo_pedido FROM pedidos;

-- D7. COUNT(DISTINCT): cuántos clientes distintos compraron
SELECT COUNT(DISTINCT cliente_id) AS clientes_que_compraron FROM pedidos;

SELECT r.nombre,
       ROUND(SUM(ri.cantidad * mp.costo_unitario), 2)                  AS costo_receta,
       ROUND(SUM(ri.cantidad * mp.costo_unitario) / r.rendimiento, 2)  AS costo_por_unidad,
       r.precio_venta,
       ROUND(r.precio_venta - SUM(ri.cantidad * mp.costo_unitario), 2) AS ganancia_bruta
FROM recetas r
JOIN receta_ingredientes ri ON ri.receta_id = r.id
JOIN materias_primas mp     ON ri.materia_prima_id = mp.id
GROUP BY r.id, r.nombre, r.rendimiento, r.precio_venta
ORDER BY ganancia_bruta DESC;
